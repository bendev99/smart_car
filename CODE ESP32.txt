#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>

const char* ssid = "ESP32 AP";
const char* password = "1234567890";

WebServer server(80);
Servo servo;

const int SERVO_PIN = 13;
int currentAngle = 90;

void setup() {
  Serial.begin(115200);

  // Configuration du Servomoteur
  servo.attach(SERVO_PIN);
  servo.write(currentAngle);

  // Démarrage du Wi-Fi en mode Point d'Accès
  WiFi.softAP(ssid, password);
  Serial.println("------------------------------------------------");
  Serial.print("Connectez-vous au Wi-Fi : ");
  Serial.println(ssid);
  Serial.print("Mot de passe : ");
  Serial.println(password);
  Serial.print("IP de l'ESP32 : ");
  Serial.println(WiFi.softAPIP().toString()); // 192.168.4.1
  Serial.println("------------------------------------------------");

  // Route HTTP pour la direction
  server.on("/cmd/steer", HTTP_GET, []() {
    if (server.hasArg("angle")) {
      int newAngle = server.arg("angle").toInt();
      // Sécurité : on limite l'angle entre 0 et 180 degrés
      currentAngle = constrain(newAngle, 0, 180); 
      servo.write(currentAngle);
      
      Serial.print("Nouvel angle du servo : ");
      Serial.println(currentAngle);
      
      server.send(200, "text/plain", "Direction: " + String(currentAngle) + " degres");
    } else {
      server.send(400, "text/plain", "Erreur: argument 'angle' manquant");
    }
  });

  server.begin();
}

void loop() {
  // Cette fonction doit rester libre pour écouter les requêtes HTTP en temps réel
  server.handleClient();
}

#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>

const char* ssid = "ESP32 AP";
const char* password = "1234567890";

WebServer server(80);
Servo servo;

const int IN1 = 19;
const int IN2 = 18;
const int ENA = 21;

const int SERVO_PIN = 13;
int currentAngle = 90;

// Configuration du PMW
const int freq = 5000;
const int resolution = 8;

void stopMotor() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  ledcWrite(ENA, 0);
}

void setMotorSpeed(int speed) {
  if (speed > 0) {
    // Marche avant
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
    ledcWrite(ENA, speed);
  } else if (speed < 0) {
    // Marche arrière (on inverse les broches et on rend la vitesse positive pour le PWM)
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, HIGH);
    ledcWrite(ENA, -speed);
  } else {
    stopMotor();
  }
}

void setup() {
  Serial.begin(115200);

  // Configuration du Servomoteur
  servo.attach(SERVO_PIN);
  servo.write(currentAngle);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  ledcAttach(ENA, freq, resolution);
  
  stopMotor();

  // Démarrage du Wi-Fi en mode Point d'Accès
  WiFi.softAP(ssid, password);
  Serial.println("------------------------------------------------");
  Serial.print("Connectez-vous au Wi-Fi : ");
  Serial.println(ssid);
  Serial.print("Mot de passe : ");
  Serial.println(password);
  Serial.print("IP de l'ESP32 : ");
  Serial.println(WiFi.softAPIP().toString()); // 192.168.4.1
  Serial.println("------------------------------------------------");

  // Route HTTP pour la direction
  server.on("/cmd/steer", HTTP_GET, []() {
    if (server.hasArg("angle")) {
      int newAngle = server.arg("angle").toInt();
      // Sécurité : on limite l'angle entre 0 et 180 degrés
      currentAngle = constrain(newAngle, 0, 180); 
      servo.write(currentAngle);
      
      Serial.print("Nouvel angle du servo : ");
      Serial.println(currentAngle);
      
      server.send(200, "text/plain", "Direction: " + String(currentAngle) + " degres");
    } else {
      server.send(400, "text/plain", "Erreur: argument 'angle' manquant");
    }
  });

  // Marche Avant
  server.on("/cmd/forward", HTTP_GET, []() {
    setMotorSpeed(200); // Vitesse à 200/255 (ajustez entre 100 et 255 selon vos besoins)
    server.send(200, "text/plain", "Marche Avant");
  });

  // Marche Arrière
  server.on("/cmd/reverse", HTTP_GET, []() {
    setMotorSpeed(-200); // Vitesse à -200/255
    server.send(200, "text/plain", "Marche Arriere");
  });

  // Arrêt d'urgence / Relâchement
  server.on("/cmd/stop", HTTP_GET, []() {
    stopMotor();
    server.send(200, "text/plain", "Arret Moteur");
  });

  server.begin();
}

void loop() {
  // Cette fonction doit rester libre pour écouter les requêtes HTTP en temps réel
  server.handleClient();
}

class CarControlApp extends StatelessWidget {
  const CarControlApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData.dark().copyWith(

    );
  }
}

class SteeringWheelScreen extends StatefulWidget {
  const SteeringWheelScreen({super.key});

  @override
  State<SteeringWheelScreen> createState() => _SteeringWheelScreenState();
}

class _SteeringWheelScreenState extends State<SteeringWheelScreen>
    with SingleTickerProviderStateMixin {
  final String espIpAddress = '192.168.4.1';

  // Angle visuel du volant : -90.0 (gauche max) à 90.0 (droite max). 0.0 = centre.
  double _visualAngle = 0.0;
  double _lastSentAngle = 90.0; // Pour éviter d'envoyer trop de requêtes HTTP
  String _statusMessage = "Prêt";

  // Contrôleur d'animation pour l'effet de retour au centre (auto-centrage)
  late AnimationController _animationController;
  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();
    _animationController = AnimationController(
      duration: const Duration(milliseconds: 300),
      vsync: this,
    );
    _animation =
        Tween<double>(begin: 0.0, end: 0.0).animate(_animationController)
          ..addListener(() {
            setState(() {
              _visualAngle = _animation.value;
            });
          });
  }

  @override
  void dispose() {
    _animationController.dispose();
    super.dispose();
  }

  Future<void> sendCommand(String endpoint) async {
    try {
      final response = await http
          .get(Uri.parse('http://$espIpAddress/cmd/$endpoint'))
          .timeout(const Duration(seconds: 2));

      if (response.statusCode == 200) {
        setState(() => _statusMessage = response.body);
      }
    } catch (e) {
      setState(() => _statusMessage = "Erreur de connexion !");
    }
  }

  void _onPanUpdate(DragUpdateDetails details) {
    setState(() {
      // La sensibilité détermine à quel point le volant tourne vite (ajustez si besoin)
      double sensitivity = 1.5;
      _visualAngle += details.delta.dx * sensitivity;

      // Limite la rotation visuelle entre -90 et 90 degrés
      _visualAngle = _visualAngle.clamp(-90.0, 90.0);
    });

    // Calcul de l'angle réel du servo (0 à 180) basé sur l'angle visuel (-90 à 90)
    double currentServoAngle = 90.0 + _visualAngle;

    // On envoie la commande HTTP seulement si l'angle a changé de plus de 5 degrés
    // Cela évite de saturer l'ESP32 avec des requêtes inutiles
    if ((currentServoAngle - _lastSentAngle).abs() > 5.0) {
      _lastSentAngle = currentServoAngle;
      sendCommand('steer?angle=${currentServoAngle.toInt()}');
    }
  }

  void _onPanEnd(DragEndDetails details) {
    // Effet d'auto-centrage : le volant revient tout seul au centre
    _animation = Tween<double>(begin: _visualAngle, end: 0.0).animate(
      CurvedAnimation(parent: _animationController, curve: Curves.easeOut),
    );
    _animationController.forward(from: 0.0);

    // On remet le servo au centre (90 degrés)
    _lastSentAngle = 90.0;
    sendCommand('steer?angle=90');
    setState(() => _statusMessage = "Recentrage...");
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            // Indicateur de statut
            Text(
              _statusMessage,
              style: const TextStyle(
                color: Colors.cyanAccent,
                fontSize: 18,
                fontWeight: FontWeight.w500,
              ),
            ),
            const SizedBox(height: 40),

            // --- LE VOLANT INTERACTIF ---
            GestureDetector(
              onPanUpdate: _onPanUpdate,
              onPanEnd: _onPanEnd,
              child: Transform.rotate(
                angle:
                    _visualAngle *
                    (pi / 180), // Conversion degrés -> radians pour la rotation
                child: Container(
                  width: 220,
                  height: 220,
                  decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    gradient: const RadialGradient(
                      colors: [Color(0xFF333333), Color(0xFF111111)],
                    ),
                    border: Border.all(
                      color: Colors.cyanAccent.withValues(alpha: 0.5),
                      width: 4,
                    ),
                    boxShadow: [
                      BoxShadow(
                        color: Colors.cyanAccent.withValues(alpha: 0.2),
                        blurRadius: 20,
                        spreadRadius: 2,
                      ),
                    ],
                  ),
                  child: Stack(
                    alignment: Alignment.center,
                    children: [
                      // Marqueur central (logo ou centre du volant)
                      Container(
                        width: 60,
                        height: 60,
                        decoration: BoxDecoration(
                          shape: BoxShape.circle,
                          color: Colors.black,
                          border: Border.all(color: Colors.white24, width: 2),
                        ),
                        child: const Icon(
                          Icons.directions_car,
                          color: Colors.cyanAccent,
                          size: 30,
                        ),
                      ),
                      // Marqueur supérieur pour voir la rotation clairement
                      Positioned(
                        top: 15,
                        child: Container(
                          width: 8,
                          height: 25,
                          decoration: BoxDecoration(
                            color: Colors.redAccent,
                            borderRadius: BorderRadius.circular(4),
                          ),
                        ),
                      ),
                      // Rayons du volant (décoration)
                      Positioned(
                        left: 20,
                        right: 20,
                        child: Container(
                          height: 12,
                          decoration: BoxDecoration(
                            color: Colors.grey.shade800,
                            borderRadius: BorderRadius.circular(6),
                          ),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ),

            const SizedBox(height: 30),

            // Instructions
            const Text(
              "Glissez pour tourner • Relâchez pour recentrer",
              style: TextStyle(
                color: Colors.white54,
                fontSize: 14,
                fontStyle: FontStyle.italic,
              ),
            ),
          ],
        ),
      ),
    );
  }
}

// --- WIDGET PERSONNALISÉ POUR LES BOUTONS MOTEUR ---
class MotorButton extends StatefulWidget {
  final String label;
  final IconData icon;
  final Color color;
  final String commandOn;
  final String commandOff;
  final Function(String) onSendCommand;

  const MotorButton({
    super.key,
    required this.label,
    required this.icon,
    required this.color,
    required this.commandOn,
    required this.commandOff,
    required this.onSendCommand,
  });

  @override
  State<MotorButton> createState() => _MotorButtonState();
}

class _MotorButtonState extends State<MotorButton> {
  bool _isPressed = false;

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      // Quand on appuie dessus
      onTapDown: (_) {
        setState(() => _isPressed = true);
        widget.onSendCommand(widget.commandOn);
      },
      // Quand on relâche le doigt
      onTapUp: (_) {
        setState(() => _isPressed = false);
        widget.onSendCommand(widget.commandOff);
      },
      // Quand on glisse le doigt hors du bouton en restant appuyé (sécurité)
      onTapCancel: () {
        setState(() => _isPressed = false);
        widget.onSendCommand(widget.commandOff);
      },
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 100),
        width: 130,
        height: 130,
        decoration: BoxDecoration(
          color: _isPressed ? widget.color.withOpacity(0.8) : widget.color,
          borderRadius: BorderRadius.circular(24),
          boxShadow: _isPressed
              ? [] // Pas d'ombre quand appuyé (effet d'enfoncement)
              : [
                  BoxShadow(
                    color: widget.color.withOpacity(0.4),
                    blurRadius: 15,
                    offset: const Offset(0, 5),
                  ),
                ],
        ),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(widget.icon, color: Colors.white, size: 40),
            const SizedBox(height: 8),
            Text(
              widget.label,
              style: const TextStyle(
                color: Colors.white,
                fontWeight: FontWeight.bold,
                fontSize: 16,
              ),
            ),
          ],
        ),
      ),
    );
  }
}
